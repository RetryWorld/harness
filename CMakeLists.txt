# Harness kernel — C++ build.
#
# MUST configure and build standalone from a checkout of the monorepo root
# (schemas/gen/cpp is read directly; nothing in edge/ regenerates it). No
# dependency reaches outside schemas/ and edge/.
#
# schemas/gen/cpp is gitignored generated output, not committed source (see
# schemas/.gitignore), so a fresh clone must run codegen before configuring:
# `buf generate` in schemas/, or scripts/install_harness.sh, which does that
# first and then runs exactly this build.
#
# Protobuf is a required dependency now: profile.proto is the single source of
# truth for the Harness Profile (build brief §3), and nothing in edge/ hand-
# rolls a parallel definition of a generated message.

cmake_minimum_required(VERSION 3.22)

project(harness_kernel
        VERSION 0.3.4
        DESCRIPTION "Physical AI harness kernel (v2 — schema-driven)"
        LANGUAGES CXX C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# Hidden by default, project-wide. Only the `extern "C"` surface (marked
# HK_API/HKC_API) is exported; a leaked C++ symbol ties consumers to our
# compiler, our standard library and our exact protobuf version -- which is
# precisely the coupling the C ABI exists to prevent.
#
# This MUST be global rather than a property on the shared library. libharness_
# kernel.so is assembled from $<TARGET_OBJECTS:...> of OBJECT libraries, and
# those objects are compiled by THEIR target's settings, not the consuming
# one -- so CXX_VISIBILITY_PRESET on harness_kernel alone silently did nothing
# and the library exported 555 harness::v1::* and protobuf symbols alongside
# the 7 it meant to. scripts/isolation_check.sh asserts the surface.
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

set(HARNESS_SCHEMAS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../schemas")
if(IS_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/schemas")
  # The downstream release repository vendors schemas/ inside its root while
  # the monorepo keeps schemas/ beside edge/.
  set(HARNESS_SCHEMAS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/schemas")
endif()
set(HARNESS_SCHEMAS_GEN_CPP "${HARNESS_SCHEMAS_DIR}/gen/cpp" CACHE PATH
    "Path to schemas/gen/cpp, buf-generated. Never regenerated by this build.")

# Fail here, with the command that fixes it, rather than 200 lines later on a
# missing .pb.cc. gen/ is generated output: an untouched clone has none.
if(NOT EXISTS "${HARNESS_SCHEMAS_GEN_CPP}/harness/v1/profile.pb.cc")
  message(FATAL_ERROR
    "No protobuf gencode at ${HARNESS_SCHEMAS_GEN_CPP}.\n"
    "schemas/gen/ is generated per-machine, not committed. Generate it first:\n"
    "  (cd <repo>/schemas && npm ci && buf generate)\n"
    "or run <repo>/scripts/install_harness.sh, which generates, builds and installs.")
endif()

# CONFIG mode (protobufConfig.cmake) only exists for protobuf builds that
# install a modern CMake package — e.g. Homebrew, vcpkg, or a from-source
# install. Ubuntu/Debian's apt libprotobuf-dev (the reference dev
# environment, §2) ships pkg-config files only, no CMake config, so CONFIG
# mode always fails there. Try it first (it is the more complete package
# when present), then fall back to CMake's own bundled FindProtobuf module,
# which locates apt's package by header/library search and still defines
# the same protobuf::libprotobuf imported target every target below links.
find_package(protobuf CONFIG QUIET)
if(NOT protobuf_FOUND)
  find_package(Protobuf REQUIRED)
endif()

# ---------------------------------------------------------------------------
# Compile options. See build brief §6 constraint 5: determinism over speed.
# -ffp-contract=off and -fno-fast-math are load-bearing for the conformance
# and determinism tests, not style.
# ---------------------------------------------------------------------------
add_library(harness_flags INTERFACE)
target_compile_options(harness_flags INTERFACE
  -ffp-contract=off
  -fno-fast-math
  -Wall -Wextra -Wpedantic -Werror
  -Wconversion -Wsign-conversion
  -Wshadow -Wnon-virtual-dtor -Wold-style-cast
  -Wcast-align -Wunused -Woverloaded-virtual
  -Wnull-dereference -Wdouble-promotion -Wformat=2
)

option(HARNESS_SANITIZE "Build with ASan+UBSan (tests and CI, never releases)" OFF)
if(HARNESS_SANITIZE)
  target_compile_options(harness_flags INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
  target_link_options(harness_flags INTERFACE -fsanitize=address,undefined)
endif()

# ---------------------------------------------------------------------------
# Generated protobuf sources. Produced by `buf generate` in schemas/, read
# from schemas/gen/cpp; never generated by this build. Warnings are silenced
# for generated code only — Werror on our own sources stays intact.
# ---------------------------------------------------------------------------
add_library(harness_proto OBJECT
  ${HARNESS_SCHEMAS_GEN_CPP}/harness/v1/enforcement.pb.cc
  ${HARNESS_SCHEMAS_GEN_CPP}/harness/v1/provenance.pb.cc
  ${HARNESS_SCHEMAS_GEN_CPP}/harness/v1/critic.pb.cc
  ${HARNESS_SCHEMAS_GEN_CPP}/harness/v1/profile.pb.cc
)
target_include_directories(harness_proto PUBLIC ${HARNESS_SCHEMAS_GEN_CPP})
target_link_libraries(harness_proto PUBLIC protobuf::libprotobuf)
target_compile_options(harness_proto PRIVATE -w)

# ---------------------------------------------------------------------------
# Profile loader + validator (build brief §3, §4.4). Hand-written: the
# structural constraints C1-C11 cannot be expressed in protobuf.
# ---------------------------------------------------------------------------
find_package(yaml-cpp CONFIG REQUIRED)

add_library(harness_profile_lib OBJECT
  src/profile/loader.cpp
  src/profile/validator.cpp
)
target_include_directories(harness_profile_lib PUBLIC include PRIVATE src)
target_link_libraries(harness_profile_lib PUBLIC harness_proto PRIVATE yaml-cpp::yaml-cpp harness_flags)

# ---------------------------------------------------------------------------
# Kernel core: Projection, Isolation accounting, two-band Transfer, the
# alloc-free event codec. Build brief §6: no heap allocation, no locks, no
# I/O, no exceptions escaping hk_gate.
# ---------------------------------------------------------------------------
add_library(harness_kernel_core OBJECT
  src/kernel/profile_runtime.cpp
  src/kernel/kernel_gate.cpp
  src/kernel/cabi.cpp
)
target_include_directories(harness_kernel_core PUBLIC include PRIVATE src)
target_link_libraries(harness_kernel_core PUBLIC harness_proto PRIVATE harness_profile_lib harness_flags)

add_library(harness_kernel SHARED $<TARGET_OBJECTS:harness_kernel_core> $<TARGET_OBJECTS:harness_proto>
                                   $<TARGET_OBJECTS:harness_profile_lib>)
target_include_directories(harness_kernel PUBLIC include)
target_link_libraries(harness_kernel PRIVATE protobuf::libprotobuf yaml-cpp::yaml-cpp harness_flags)
set_target_properties(harness_kernel PROPERTIES
  OUTPUT_NAME harness_kernel
  VERSION ${PROJECT_VERSION}
  SOVERSION 1  # tracks HK_ABI_VERSION, not the release
  CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON
)

add_library(harness_kernel_static STATIC $<TARGET_OBJECTS:harness_kernel_core> $<TARGET_OBJECTS:harness_proto>
                                          $<TARGET_OBJECTS:harness_profile_lib>)
target_include_directories(harness_kernel_static PUBLIC include)
target_link_libraries(harness_kernel_static PUBLIC protobuf::libprotobuf yaml-cpp::yaml-cpp)
set_target_properties(harness_kernel_static PROPERTIES OUTPUT_NAME harness_kernel)

# ---------------------------------------------------------------------------
# Critic ABI: the two deliberately trivial reference critics, each its own
# shared library so the ABI story (a critic is a compiled artifact loaded
# independently of the kernel) is real rather than simulated by static
# linkage. See build brief §7.
# ---------------------------------------------------------------------------
add_library(threshold_critic SHARED src/critic/threshold_critic.cpp)
target_include_directories(threshold_critic PRIVATE include src)
target_link_libraries(threshold_critic PRIVATE harness_flags)

add_library(stall_critic SHARED src/critic/stall_critic.cpp)
target_include_directories(stall_critic PRIVATE include src)
target_link_libraries(stall_critic PRIVATE harness_flags)
set_target_properties(stall_critic PROPERTIES CXX_VISIBILITY_PRESET hidden)
